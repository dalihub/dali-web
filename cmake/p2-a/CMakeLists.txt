cmake_minimum_required(VERSION 3.18)
project(dali-web-gles-diagnostic LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Use emcmake")
endif()
get_filename_component(DALI_WEB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(STATIC ON CACHE BOOL "Static Core" FORCE)
set(ENABLE_PKG_CONFIGURE OFF CACHE BOOL "No source outputs" FORCE)
set(ENABLE_LINK_TEST OFF CACHE BOOL "No mock helpers" FORCE)
set(ENABLE_TIZEN_NAMESPACE OFF CACHE BOOL "Dali namespace" FORCE)
set(DALI_GENERATED_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated/include")
add_compile_options(-pthread)
add_link_options(-pthread)
add_subdirectory("${DALI_WEB_ROOT}/libs/dali-core/build/tizen" core)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/sources.txt" diagnostic_sources)
list(TRANSFORM diagnostic_sources PREPEND "${DALI_WEB_ROOT}/")
set(cachePath "/diagnostic-only")
configure_file("${DALI_WEB_ROOT}/libs/dali-adaptor/build/tizen/shader-cache-path.in" "${CMAKE_CURRENT_BINARY_DIR}/shader-cache-path.cpp" @ONLY)
add_library(gles-diagnostic OBJECT "${CMAKE_CURRENT_BINARY_DIR}/shader-cache-path.cpp" ${diagnostic_sources} "${DALI_WEB_ROOT}/tests/probes/gles-link-diagnostic.cpp")
target_compile_features(gles-diagnostic PRIVATE cxx_std_17)
target_compile_definitions(gles-diagnostic PRIVATE DEBUG_ENABLED)
target_include_directories(gles-diagnostic PRIVATE "${DALI_GENERATED_INCLUDE_DIR}" "${DALI_WEB_ROOT}/libs/dali-core" "${DALI_WEB_ROOT}/libs/dali-adaptor")
