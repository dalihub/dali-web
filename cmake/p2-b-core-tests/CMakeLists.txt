cmake_minimum_required(VERSION 3.18)
project(dali-web-core-failure-tests LANGUAGES C CXX)
get_filename_component(DALI_WEB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_compile_options(-pthread)
add_link_options(-pthread)
add_subdirectory("${DALI_WEB_ROOT}/cmake" probes)
get_target_property(test_sources linker-test SOURCES)
list(REMOVE_ITEM test_sources linker-test.cpp)
add_executable(core-buffer-failure "${DALI_WEB_ROOT}/tests/unit/core-buffer-failure.cpp" ${test_sources})
target_compile_features(core-buffer-failure PRIVATE cxx_std_17)
target_include_directories(core-buffer-failure PRIVATE "${DALI_WEB_ROOT}/libs/dali-core/automated-tests/src/dali/dali-test-suite-utils")
target_link_libraries(core-buffer-failure PRIVATE dali2-core)
target_link_options(core-buffer-failure PRIVATE -sASSERTIONS=1 -sEXIT_RUNTIME=1 -sENVIRONMENT=node -sERROR_ON_UNDEFINED_SYMBOLS=1)
